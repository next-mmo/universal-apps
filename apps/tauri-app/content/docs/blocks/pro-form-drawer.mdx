## Anatomy

```tsx
import { ProFormDrawer } from '@package/pro/form-drawer';
import { Button } from '@package/ui/button';

export function Example() {
  return (
    <ProFormDrawer
      title="Create Workspace"
      description="Configure workspace details and settings."
      trigger={<Button>New Workspace</Button>}
      schema={[
        {
          title: 'General',
          fields: [
            { name: 'name', label: 'Workspace Name', type: 'text', required: true },
            { name: 'slug', label: 'Slug', type: 'text', required: true },
          ],
        },
      ]}
      defaultValues={{ name: '', slug: '' }}
      onSubmit={async (values) => {
        console.log('Submitted', values);
        return true; // Closes drawer on success
      }}
    />
  );
}
```
